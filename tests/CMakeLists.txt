cmake_minimum_required(VERSION 3.16)
project(OBSMirrorVulkanTests LANGUAGES CXX)
find_package(Vulkan REQUIRED)
enable_testing()
add_executable(vulkan_readback_test vulkan_readback_test.cpp
    ../XR_APILAYER_NOVENDOR_OBSMirror/vulkan_readback.cpp)
target_include_directories(vulkan_readback_test PRIVATE ../XR_APILAYER_NOVENDOR_OBSMirror)
target_compile_features(vulkan_readback_test PRIVATE cxx_std_17)
target_link_libraries(vulkan_readback_test PRIVATE Vulkan::Vulkan)
if(MSVC)
    target_compile_options(vulkan_readback_test PRIVATE /W4)
else()
    target_compile_options(vulkan_readback_test PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
endif()
add_test(NAME vulkan_readback COMMAND vulkan_readback_test)
if(TARGET XR_APILAYER_NOVENDOR_OBSMirror)
    add_executable(linux_layer_test linux_layer_test.cpp
        ../XR_APILAYER_NOVENDOR_OBSMirror/vulkan_readback.cpp)
    target_include_directories(linux_layer_test PRIVATE "${OPENXR_INCLUDE_DIR}" ../XR_APILAYER_NOVENDOR_OBSMirror)
    target_link_libraries(linux_layer_test PRIVATE XR_APILAYER_NOVENDOR_OBSMirror mirror_ipc Vulkan::Vulkan)
    target_compile_options(linux_layer_test PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    add_test(NAME linux_layer COMMAND linux_layer_test)
    add_executable(obs_module_test obs_module_test.cpp)
    target_include_directories(obs_module_test PRIVATE "${OBS_INCLUDE_DIR}")
    if(SIMDE_INCLUDE_DIR)
        target_include_directories(obs_module_test SYSTEM PRIVATE "${SIMDE_INCLUDE_DIR}")
    endif()
    target_link_libraries(obs_module_test PRIVATE "${OBS_LIBRARY}")
    add_dependencies(obs_module_test openxr-vulkan-mirror)
    add_test(NAME obs_module COMMAND obs_module_test $<TARGET_FILE:openxr-vulkan-mirror>)
endif()
