cmake_minimum_required(VERSION 3.16)
project(OpenXROBSMirror VERSION 0.4.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use the Visual Studio solution for Windows builds")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
find_path(OBS_INCLUDE_DIR obs-module.h PATH_SUFFIXES obs REQUIRED)
find_path(SIMDE_INCLUDE_DIR simde/x86/sse2.h)
find_library(OBS_LIBRARY obs REQUIRED)
set(OPENXR_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/external/OpenXR-SDK/include" CACHE PATH "OpenXR headers")
if(NOT EXISTS "${OPENXR_INCLUDE_DIR}/openxr/openxr_loader_negotiation.h")
    message(FATAL_ERROR "Initialize external/OpenXR-SDK or set OPENXR_INCLUDE_DIR")
endif()
add_library(mirror_ipc STATIC linux/frame_ipc.cpp)
target_include_directories(mirror_ipc PUBLIC linux)
add_library(XR_APILAYER_NOVENDOR_OBSMirror SHARED linux/layer.cpp
    XR_APILAYER_NOVENDOR_OBSMirror/vulkan_readback.cpp)
target_include_directories(XR_APILAYER_NOVENDOR_OBSMirror PRIVATE
    "${OPENXR_INCLUDE_DIR}" XR_APILAYER_NOVENDOR_OBSMirror)
target_link_libraries(XR_APILAYER_NOVENDOR_OBSMirror PRIVATE mirror_ipc Vulkan::Vulkan Threads::Threads)
set_target_properties(XR_APILAYER_NOVENDOR_OBSMirror PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_library(openxr-vulkan-mirror MODULE linux/obs_plugin.cpp)
target_include_directories(openxr-vulkan-mirror PRIVATE "${OBS_INCLUDE_DIR}")
if(SIMDE_INCLUDE_DIR)
    target_include_directories(openxr-vulkan-mirror SYSTEM PRIVATE "${SIMDE_INCLUDE_DIR}")
endif()
target_link_libraries(openxr-vulkan-mirror PRIVATE mirror_ipc "${OBS_LIBRARY}" Threads::Threads)
set_target_properties(openxr-vulkan-mirror PROPERTIES PREFIX "")
foreach(target mirror_ipc XR_APILAYER_NOVENDOR_OBSMirror openxr-vulkan-mirror)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
install(TARGETS XR_APILAYER_NOVENDOR_OBSMirror openxr-vulkan-mirror LIBRARY DESTINATION lib)
install(PROGRAMS scripts/Install-Linux.sh scripts/Uninstall-Linux.sh DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY DESTINATION .)
install(FILES docs/LINUX.md docs/VULKAN.md DESTINATION docs)
